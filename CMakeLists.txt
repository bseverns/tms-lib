cmake_minimum_required(VERSION 3.16)
project(tms-lib VERSION 0.1.0 LANGUAGES CXX)
add_library(tms INTERFACE)
add_library(tms::tms ALIAS tms)
target_include_directories(tms INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/lib")
target_compile_features(tms INTERFACE cxx_std_17)
option(TMS_BUILD_EXAMPLES "Build the portable teaching example" ON)
option(TMS_BUILD_TESTS "Build portable behavior and header checks" ON)
if(TMS_BUILD_EXAMPLES)
  add_executable(tms_patch examples/native/patch.cpp)
  target_link_libraries(tms_patch PRIVATE tms::tms)
  add_executable(tms_drum examples/native/drum_voice.cpp)
  target_link_libraries(tms_drum PRIVATE tms::tms)
  add_executable(tms_sample examples/native/sample_playback.cpp)
  target_link_libraries(tms_sample PRIVATE tms::tms)
  add_executable(tms_convolution examples/native/convolution.cpp)
  target_link_libraries(tms_convolution PRIVATE tms::tms)
  add_executable(tms_granular examples/native/granular.cpp)
  target_link_libraries(tms_granular PRIVATE tms::tms)
  add_executable(tms_frequency_shift examples/native/frequency_shift.cpp)
  target_link_libraries(tms_frequency_shift PRIVATE tms::tms)
  add_executable(tms_spectral examples/native/spectral_hold.cpp)
  target_link_libraries(tms_spectral PRIVATE tms::tms)
endif()
if(TMS_BUILD_TESTS)
  enable_testing()
  add_executable(tms_tests tests/behavior.cpp)
  target_link_libraries(tms_tests PRIVATE tms::tms)
  add_test(NAME behavior COMMAND tms_tests)
  add_executable(tms_synthesis_tests tests/synthesis.cpp)
  target_link_libraries(tms_synthesis_tests PRIVATE tms::tms)
  add_test(NAME synthesis COMMAND tms_synthesis_tests)
  add_executable(tms_sample_tests tests/sample.cpp)
  target_link_libraries(tms_sample_tests PRIVATE tms::tms)
  add_test(NAME sample COMMAND tms_sample_tests)
  add_executable(tms_fir_tests tests/fir.cpp)
  target_link_libraries(tms_fir_tests PRIVATE tms::tms)
  add_test(NAME fir COMMAND tms_fir_tests)
  add_executable(tms_granular_tests tests/granular.cpp)
  target_link_libraries(tms_granular_tests PRIVATE tms::tms)
  add_test(NAME granular COMMAND tms_granular_tests)
  add_executable(tms_frequency_shift_tests tests/frequency_shifter.cpp)
  target_link_libraries(tms_frequency_shift_tests PRIVATE tms::tms)
  add_test(NAME frequency_shift COMMAND tms_frequency_shift_tests)
  add_executable(tms_spectral_tests tests/spectral.cpp)
  target_link_libraries(tms_spectral_tests PRIVATE tms::tms)
  add_test(NAME spectral COMMAND tms_spectral_tests)
  add_executable(tms_audio_lab_tests tests/audio_lab.cpp)
  target_link_libraries(tms_audio_lab_tests PRIVATE tms::tms)
  add_test(NAME audio_lab COMMAND tms_audio_lab_tests)
  file(GLOB_RECURSE TMS_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/lib/tms/*.h")
  foreach(header IN LISTS TMS_HEADERS)
    file(RELATIVE_PATH include_path "${CMAKE_CURRENT_SOURCE_DIR}/lib" "${header}")
    string(MAKE_C_IDENTIFIER "${include_path}" unit)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${unit}.cpp" "#include \"${include_path}\"\n")
    add_library("header_${unit}" OBJECT "${CMAKE_CURRENT_BINARY_DIR}/${unit}.cpp")
    target_link_libraries("header_${unit}" PRIVATE tms::tms)
  endforeach()
endif()
